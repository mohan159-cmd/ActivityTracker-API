CREATE DATABASE EmployeeDB;

USE EmployeeDB;

CREATE TABLE Departments (
    DepartmentID INT PRIMARY KEY IDENTITY(1,1),  -- IDENTITY used for auto-increment
    DepartmentName VARCHAR(100),
    ManagerID INT NULL  -- Assuming a manager can be an employee (FK can be set later)
);

CREATE TABLE Positions (
    PositionID INT PRIMARY KEY IDENTITY(1,1),  -- IDENTITY used for auto-increment
    PositionName VARCHAR(100),
    DepartmentID INT FOREIGN KEY REFERENCES Departments(DepartmentID)  -- Foreign key to Departments table
);

CREATE TABLE Employees (
    EmployeeID INT PRIMARY KEY IDENTITY(1,1),  -- IDENTITY used for auto-increment
    FirstName VARCHAR(50),
    LastName VARCHAR(50),
    DateOfBirth DATE,
    Gender VARCHAR(10),  -- SQL Server doesn't support ENUM directly
    HireDate DATE,
    Position VARCHAR(100),
    Department VARCHAR(100),
    Salary DECIMAL(10, 2),
    Email VARCHAR(100),
    PhoneNumber VARCHAR(15),
    Address VARCHAR(255)
);
